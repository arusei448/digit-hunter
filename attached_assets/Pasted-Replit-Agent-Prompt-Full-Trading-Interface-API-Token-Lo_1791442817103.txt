            );

            const data = await response.json();

            if (!response.ok) {
                res.writeHead(response.status, { 'Content-Type': 'application/json' });
                res.end(JSON.stringify(data));
                return;
            }

            res.writeHead(200, { 'Content-Type': 'application/json' });
            res.end(JSON.stringify(data));
        } catch (err) {
            res.writeHead(500, { 'Content-Type': 'application/json' });
            res.end(JSON.stringify({ error: err.message }));
        }
    });
    return;
}

javascript
let authenticatedWs = null;
let currentAccount = null;
let currentBalance = 0;

// Connect to authenticated WebSocket
async function connectTrading(token, accountId) {
    const res = await fetch('/api/otp', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ token, accountId })
    });
    const data = await res.json();
    if (!data.data || !data.data.url) {
        throw new Error('Failed to get OTP URL');
    }
    authenticatedWs = new WebSocket(data.data.url);
    return new Promise((resolve, reject) => {
        authenticatedWs.onopen = () => {
            // Subscribe to balance
            authenticatedWs.send(JSON.stringify({ balance: 1, subscribe: 1 }));
            // Get portfolio
            authenticatedWs.send(JSON.stringify({ portfolio: 1 }));
            resolve();
        };
        authenticatedWs.onerror = reject;
    });
}

// Request a proposal
function requestProposal(params) {
    authenticatedWs.send(JSON.stringify({
        proposal: 1,
        amount: params.amount,
        basis: 'stake',
        contract_type: params.contractType,  // 'CALL' or 'PUT'
        currency: 'USD',
        underlying_symbol: params.symbol,
        duration: params.duration,
        duration_unit: params.durationUnit  // 't' for ticks, 's' for seconds
    }));
}

// Buy a contract
function buyContract(proposalId, askPrice) {
    authenticatedWs.send(JSON.stringify({
        buy: proposalId,
        price: askPrice
    }));
}

// Monitor a contract
function monitorContract(contractId) {
    authenticatedWs.send(JSON.stringify({
        proposal_open_contract: 1,
        contract_id: contractId,
        subscribe: 1
    }));
}

// Sell early
function sellContract(contractId) {
    authenticatedWs.send(JSON.stringify({
        sell: contractId,
        price: 0    // 0 = sell at market
    }));
}
Message handler:

javascript
authenticatedWs.onmessage = (event) => {
    const msg = JSON.parse(event.data);

    switch (msg.msg_type) {
        case 'balance':
            currentBalance = msg.balance.balance;
            updateBalanceDisplay(currentBalance);
            break;
        case 'proposal':
            // Store proposal for buy button
            currentProposal = msg.proposal;
            updatePayoutPreview(msg.proposal.payout);
            break;
        case 'buy':
            // Contract bought — start monitoring
            monitorContract(msg.buy.contract_id);
            addOpenPosition(msg.buy);
            break;
        case 'proposal_open_contract':
            updateContractStatus(msg.proposal_open_contract);
            break;
        case 'sell':
            removeOpenPosition(msg.sell.contract_id);
            break;
        case 'portfolio':
            msg.portfolio.contracts.forEach(addOpenPosition);
            break;
    }
};

Core bot loop:

javascript
let botRunning = false;
let botState = {
    round: 0,
    currentStake: 0,
    totalPnL: 0,
    wins: 0,
    losses: 0
};

async function startBot(config) {
    botRunning = true;
    botState.currentStake = config.initialStake;

    // List available strategies first
    const strategies = await listStrategies();
    // Find the selected strategy ID
    const strategyId = strategies.find(s => s.name === config.strategy).id;

    // Start the automated run
    authenticatedWs.send(JSON.stringify({
        auto_start: 1,
        contract_template: {
            contract_type: 'CALL',
            currency: 'USD',
            underlying_symbol: config.symbol,
            duration: config.duration,
            duration_unit: 't'
        },
        strategy_id: strategyId,
        strategy_parameters: {
            initial_stake: config.initialStake,
            multiplier: config.multiplier,
            profit_threshold: config.profitThreshold,
            loss_threshold: config.lossThreshold,
            max_stake: config.maxStake
        },
        subscribe: 1
    }));
}

async function listStrategies() {
    return new Promise((resolve) => {
        const reqId = Date.now();
        const handler = (event) => {
            const msg = JSON.parse(event.data);
            if (msg.req_id === reqId) {
                authenticatedWs.removeEventListener('message', handler);
                resolve(msg.auto_list_strategies.strategies);
            }
        };
        authenticatedWs.addEventListener('message', handler);
        authenticatedWs.send(JSON.stringify({
            auto_list_strategies: 1,
            req_id: reqId
        }));
    });
}

function stopBot() {
    botRunning = false;
    authenticatedWs.send(JSON.stringify({
        auto_stop: 1
    }));
}
