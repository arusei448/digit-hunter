## PART 5 — AI Analyser Page Logic

### File: `site/ai-analyser.js`

```javascript
// State
let currentStrategy = 'trend';   // 'trend' or 'reversion'
let latestAnalysis = null;

// DOM
const tpMarket = document.getElementById('tp-market');
const tpPrice = document.getElementById('tp-price');
const tpScore = document.getElementById('tp-score');
const tpRationale = document.getElementById('tp-rationale');
const marketRows = document.getElementById('market-rows');

// ---------- Strategy toggle ----------
document.querySelectorAll('.strategy-btn').forEach(btn => {
    btn.addEventListener('click', () => {
        document.querySelectorAll('.strategy-btn').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
        currentStrategy = btn.dataset.strategy;
        if (latestAnalysis) render(latestAnalysis);
    });
});

// ---------- Rationale builder ----------
function buildRationale(market) {
    const parts = [];

    if (currentStrategy === 'trend') {
        if (market.rSquared > 0.7) parts.push(`<strong>Strong trend</strong> detected (R² ${market.rSquared.toFixed(2)})`);
        else if (market.rSquared > 0.4) parts.push(`Moderate trend (R² ${market.rSquared.toFixed(2)})`);
        else parts.push(`Weak directional bias (R² ${market.rSquared.toFixed(2)})`);

        if (market.riseFall.upPct > 60) parts.push(`bullish momentum (${market.riseFall.upPct.toFixed(0)}% up-ticks)`);
        else if (market.riseFall.upPct < 40) parts.push(`bearish momentum (${(100 - market.riseFall.upPct).toFixed(0)}% down-ticks)`);
        else parts.push(`balanced tick flow (${market.riseFall.upPct.toFixed(0)}% up)`);

        if (market.hurst > 0.55) parts.push(`Hurst ${market.hurst.toFixed(2)} indicates trending behaviour`);
    } else {
        if (market.hurst < 0.45) parts.push(`Hurst ${market.hurst.toFixed(2)} suggests <strong>mean-reverting</strong> behaviour`);
        else parts.push(`Hurst ${market.hurst.toFixed(2)} — mixed regime`);

        if (market.digitDistribution.chiSq > 15) parts.push(`notable digit skew (χ² ${market.digitDistribution.chiSq.toFixed(1)})`);
        else parts.push(`digit distribution near uniform (χ² ${market.digitDistribution.chiSq.toFixed(1)})`);

        if (market.streak.currentStreak >= 4) parts.push(`current ${market.streak.currentStreak}-tick ${market.streak.direction} streak`);
    }

    return parts.join(' · ');
}

// ---------- Rendering ----------
function render(data) {
    const markets = data.markets || [];
    if (markets.length === 0) return;

    // Sort by the current strategy
    const key = currentStrategy === 'trend' ? 'trendScore' : 'reversionScore';
    const sorted = markets.slice().sort((a, b) => b[key] - a[key]);

    // Top pick
    const top = sorted[0];
    if (top) {
        tpMarket.textContent = top.name;
        tpPrice.textContent = `${top.currentPrice.toFixed(2)}  ·  ${top.tickCount} ticks analysed`;
        tpScore.textContent = `Score: ${top[key]} / 100`;
        tpRationale.innerHTML = buildRationale(top);
    }

    // Table
    marketRows.innerHTML = '';
    sorted.forEach((m, i) => {
        const tr = document.createElement('tr');

        let rankClass = '';
        if (i === 0) rankClass = 'top1';
        else if (i === 1) rankClass = 'top2';
        else if (i === 2) rankClass = 'top3';

        const score = m[key];
        const risePct = m.riseFall.upPct;
        const riseClass = risePct > 50 ? 'green' : risePct < 50 ? 'red' : '';

        tr.innerHTML = `
            <td><span class="rank-badge ${rankClass}">${i + 1}</span></td>
            <td class="market-cell">${m.name}</td>
            <td class="num-cell">${m.currentPrice.toFixed(2)}</td>
            <td>
                <span class="score-bar"><span class="score-bar-fill" style="width:${score}%"></span></span>
                <span class="score-cell">${score}</span>
            </td>
            <td class="num-cell hide-mobile">${m.rSquared.toFixed(2)}</td>
            <td class="num-cell ${riseClass} hide-mobile">${risePct.toFixed(1)}%</td>
            <td class="num-cell hide-mobile">${(m.volatility * 100).toFixed(2)}%</td>
            <td class="num-cell hide-mobile">${m.hurst.toFixed(2)}</td>
            <td class="num-cell hide-mobile">${m.streak.currentStreak} ${m.streak.direction}</td>
        `;
        marketRows.appendChild(tr);
    });
}

// ---------- SSE connection ----------
const eventSource = new EventSource('/stream/ai');

eventSource.onmessage = (event) => {
    latestAnalysis = JSON.parse(event.data);
    render(latestAnalysis);
};

eventSource.onerror = () => {
    // Keep trying — SSE auto-reconnects
};
```

Add this at the bottom of the file, after the existing SSE setup:

// ---------- AI Insight widget ----------
const aiInsightMarket = document.getElementById('ai-insight-market');
const aiInsightScore = document.getElementById('ai-insight-score');
const aiInsightNote = document.getElementById('ai-insight-note');

const aiEventSource = new EventSource('/stream/ai');

aiEventSource.onmessage = (event) => {
    const data = JSON.parse(event.data);
    const markets = data.markets || [];
    if (markets.length === 0) return;

    // Rank by trend-following score (default for this widget)
    const sorted = markets.slice().sort((a, b) => b.trendScore - a.trendScore);
    const top = sorted[0];

    aiInsightMarket.textContent = top.name;
    aiInsightScore.textContent = `${top.trendScore} / 100`;

    const parts = [];
    if (top.rSquared > 0.7) parts.push(`R² ${top.rSquared.toFixed(2)}`);
    if (top.riseFall.upPct > 60) parts.push(`${top.riseFall.upPct.toFixed(0)}% up`);
    else if (top.riseFall.upPct < 40) parts.push(`${(100 - top.riseFall.upPct).toFixed(0)}% down`);

    aiInsightNote.textContent = parts.length > 0
        ? `Top trend candidate · ${parts.join(' · ')}`
        : 'Top trend-following candidate based on current conditions.';
};
