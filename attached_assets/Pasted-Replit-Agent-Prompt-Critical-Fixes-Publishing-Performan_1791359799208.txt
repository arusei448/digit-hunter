FIX 3 — Add SSE Production Headers (Fixes "No Ticks" on Deployed Site)
Why this fixes the problem: Nginx buffers responses by default, delaying SSE events until ~16KB accumulates. This breaks real-time delivery. The X-Accel-Buffering: no header tells the proxy not to buffer for SSE responses.

index.js — Update ALL SSE endpoints
Find every res.writeHead(200, {...}) block that handles an SSE route (/stream, /stream/direction, /stream/ai) and add two headers:

javascript
res.writeHead(200, {
    'Content-Type': 'text/event-stream',
    'Cache-Control': 'no-cache, no-transform',
    'Connection': 'keep-alive',
    'X-Accel-Buffering': 'no',
    'Access-Control-Allow-Origin': '*'
});
Add 'X-Accel-Buffering': 'no' and 'no-transform' to Cache-Control. Both are critical — the first prevents Nginx from buffering, the second prevents compression middleware from breaking chunked delivery.

FIX 4 — Add SSE Heartbeat (Prevents Connection Drops in Production)
Why this fixes the problem: SSE connections through proxy chains are killed after ~100–125 seconds of idle time. A heartbeat every 30 seconds keeps the TCP connection warm and prevents buffering.

index.js — Add heartbeat interval
Add this near the bottom of index.js, alongside the other setInterval calls:

javascript
// SSE heartbeat — keeps connections alive through proxies
setInterval(() => {
    const heartbeat = ': heartbeat\n\n';
    for (const client of sseClients) {
        try { client.write(heartbeat); } catch (e) { sseClients.delete(client); }
    }
    for (const client of directionClients) {
        try { client.write(heartbeat); } catch (e) { directionClients.delete(client); }
    }
    for (const client of aiClients) {
        try { client.write(heartbeat); } catch (e) { aiClients.delete(client); }
    }
}, 30000);
The : prefix is an SSE comment — the browser ignores it, but the proxy sees data flowing and keeps the connection open.

index.js — Configure HTTP server timeouts
Find where server.listen(...) is called. Before that line, add:

javascript

server.headersTimeout = 66000;

Why: Node.js defaults to 5-second keep-alive and 60-second header timeouts. These kill SSE connections between ticks. Setting timeout = 0 disables socket timeouts for long-lived streams.

if (req.url.startsWith('/switch')) {
    const url = new URL(req.url, `http://localhost:${PORT}`);
    const newSymbol = url.searchParams.get('symbol');

    // 6. Return history directly
    res.writeHead(200, { 'Content-Type': 'application/json' });

        symbol: currentSymbol,
        history: ticks.slice(-1000)
    }));

index.js — Add req_id to fetchHistory
Replace the fetchHistory function with:

javascript
let reqIdCounter = 1;

function fetchHistory(symbol) {
    return new Promise((resolve) => {
        const reqId = reqIdCounter++;
        historyResolvers.push({ reqId, resolve });

        ws.send(JSON.stringify({
            ticks_history: symbol,
            end: 'latest',
            start: 0,
            style: 'ticks',
            count: 1000,
            subscribe: 0,

    });

Change historyResolvers from an array of functions to an array of { reqId, resolve } objects. Update the history message handler to find by req_id.

Why req_id: Without it, when multiple requests are in flight, responses can't be matched to their originating request.

FIX 8 — WebSocket Reconnection with Exponential Backoff
index.js — Replace the WebSocket connection block
Replace the existing const ws = new WebSocket(WS_URL); and all its handlers with:

javascript
let ws = null;
let reconnectAttempt = 0;
const MAX_RECONNECT_DELAY = 30000;
let currentSubscriptions = [];

function connectWebSocket() {
    ws = new WebSocket(WS_URL);

    ws.on('open', () => {
        console.log('Connected to Deriv public gateway');
        reconnectAttempt = 0;

        // Re-subscribe to all active markets
        currentSubscriptions.forEach(sub => {
            ws.send(JSON.stringify(sub));
        });

        // Fetch initial history
        fetchHistory(currentSymbol).then(ticks => {
            historicalTicks = ticks;
            console.log(`Loaded ${ticks.length} initial ticks for ${currentSymbol}`);
        });

        // Subscribe to current symbol
        const sub = { ticks: currentSymbol, subscribe: 1 };
        currentSubscriptions.push(sub);
        ws.send(JSON.stringify(sub));

        // Subscribe to all markets for AI analyser
        ALL_MARKETS.forEach(sym => {
            const s = { ticks: sym, subscribe: 1 };
            ws.send(JSON.stringify(s));
        });
    });

    ws.on('message', (data) => {
        // ... existing message handler logic stays here ...
    });

    ws.on('error', (err) => console.log('WS Error:', err.message));

    ws.on('close', () => {
        console.log('WS closed — reconnecting...');
        const delay = Math.min(1000 * Math.pow(2, reconnectAttempt), MAX_RECONNECT_DELAY);
        const jitter = Math.random() * 500;
        reconnectAttempt++;
        console.log(`Reconnecting in ${(delay + jitter).toFixed(0)}ms (attempt ${reconnectAttempt})`);
        setTimeout(connectWebSocket, delay + jitter);
    });
}

connectWebSocket();
Why: Exponential backoff with jitter (1s, 2s, 4s, 8s, capped at 30s) prevents reconnection storms and ensures the app recovers from network hiccups automatically.
