## FIX 2 — Stop Elements from Popping / Flashing on Every Tick

**Problem:** On every new tick, the digit row and stream are rebuilt with `innerHTML = ''`, causing visible flashing.

**Solution:** Build the DOM elements ONCE at startup, then update only text and classes in place.

### `site/app.js`

**Add a new `initDigitElements()` function:**

```javascript
function initDigitElements() {
    const container = document.getElementById('digits');
    container.innerHTML = '';
    for (let d = 0; d < 10; d++) {
        const wrap = document.createElement('div');
        wrap.style.cssText = 'display:flex;flex-direction:column;align-items:center;flex:1;';

        const circle = document.createElement('div');
        circle.className = 'digit-circle';
        circle.id = `digit-circle-${d}`;

        const numSpan = document.createElement('span');
        numSpan.id = `digit-num-${d}`;
        numSpan.textContent = d;

        const pctSpan = document.createElement('span');
        pctSpan.className = 'digit-percent';
        pctSpan.id = `digit-pct-${d}`;
        pctSpan.textContent = '0.0%';

        circle.appendChild(numSpan);
        circle.appendChild(pctSpan);
        wrap.appendChild(circle);

        const label = document.createElement('div');
        label.className = 'digit-label';
        label.id = `digit-label-${d}`;
        label.textContent = '';
        wrap.appendChild(label);

        container.appendChild(wrap);
    }
}
```

**Replace the entire `renderDigits()` function with:**

```javascript
function renderDigits() {
    const total = allTicks.length;
    if (total === 0) return;

    const latestDigit = allTicks[total - 1].digit;

    let hotDigit = 0, coldDigit = 0;
    let maxCount = digitCounts[0], minCount = digitCounts[0];
    for (let i = 1; i < 10; i++) {
        if (digitCounts[i] > maxCount) { maxCount = digitCounts[i]; hotDigit = i; }
        if (digitCounts[i] < minCount) { minCount = digitCounts[i]; coldDigit = i; }
    }

    for (let d = 0; d < 10; d++) {
        const circle = document.getElementById(`digit-circle-${d}`);
        const pctEl = document.getElementById(`digit-pct-${d}`);
        const labelEl = document.getElementById(`digit-label-${d}`);
        if (!circle) continue;

        // Reset state
        circle.classList.remove('latest', 'hot', 'cold');
        labelEl.classList.remove('latest', 'hot', 'cold');
        labelEl.textContent = '';

        // Apply one state per digit — latest wins, then hot, then cold
        if (d === latestDigit) {
            circle.classList.add('latest');
            labelEl.textContent = 'Latest';
            labelEl.classList.add('latest');
        } else if (d === hotDigit) {
            circle.classList.add('hot');
            labelEl.textContent = 'Most';
            labelEl.classList.add('hot');
        } else if (d === coldDigit) {
            circle.classList.add('cold');
            labelEl.textContent = 'Least';
            labelEl.classList.add('cold');
        }

        const pct = ((digitCounts[d] / total) * 100).toFixed(1);
        pctEl.textContent = pct + '%';
    }

    document.getElementById('total-ticks').textContent = total;
    document.getElementById('latest-digit').textContent = latestDigit;
    document.getElementById('hot-digit').textContent = hotDigit;
    document.getElementById('cold-digit').textContent = coldDigit;
}
```

    const quoteRaw = msg.tick.quote.toString();

        ? 'flat'
        : quoteNum > previousQuote ? 'up'
        : quoteNum < previousQuote ? 'down'
