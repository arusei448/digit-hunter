FIX 2 — Add SSE Production Headers (Critical — Fixes "No Ticks")
index.js
Find every SSE endpoint (/stream, /stream/direction, /stream/ai). For each res.writeHead(200, {...}) block, ensure the headers include:

javascript
res.writeHead(200, {
    'Content-Type': 'text/event-stream',
    'Cache-Control': 'no-cache, no-transform',
    'Connection': 'keep-alive',
    'X-Accel-Buffering': 'no',
    'Access-Control-Allow-Origin': '*'
});
The two critical additions are:

'X-Accel-Buffering': 'no' — tells Replit's Nginx proxy not to buffer SSE responses

'no-transform' in Cache-Control — prevents compression middleware from breaking chunked delivery

Why: Nginx buffers responses by default. Without X-Accel-Buffering: no, SSE events are delayed until ~16KB accumulates — which may never happen for low-frequency tick streams.

javascript
// Ping Deriv every 25 seconds to keep the connection alive
setInterval(() => {

Why: Deriv closes idle WebSocket connections after 30 seconds. Without a ping, the connection dies silently and your backend never receives ticks.

FIX 5 — WebSocket Reconnection with Exponential Backoff
index.js
Replace the WebSocket connection block (const ws = new WebSocket(WS_URL) and all its handlers) with:

javascript
let ws = null;
let reconnectAttempt = 0;
const MAX_RECONNECT_DELAY = 30000;
let currentSubscriptions = [];
let reqIdCounter = 1;

function connectWebSocket() {
    ws = new WebSocket(WS_URL);

    ws.on('open', () => {
        console.log('Connected to Deriv public gateway');
        reconnectAttempt = 0;

        // Re-subscribe to all active markets after reconnect
        currentSubscriptions.forEach(sub => {
            ws.send(JSON.stringify(sub));
        });

        // Fetch initial history for current symbol
        fetchHistory(currentSymbol).then(ticks => {
            historicalTicks = ticks;
            console.log(`Loaded ${ticks.length} initial ticks for ${currentSymbol}`);
        });

        // Subscribe to current symbol
        const sub = { ticks: currentSymbol, subscribe: 1 };
        currentSubscriptions.push(sub);
        ws.send(JSON.stringify(sub));
    });

    ws.on('message', (data) => {
        // ... existing message handler logic stays here ...
    });

    ws.on('error', (err) => console.log('WS Error:', err.message));

    ws.on('close', () => {
        console.log('WS closed — reconnecting...');
        const delay = Math.min(1000 * Math.pow(2, reconnectAttempt), MAX_RECONNECT_DELAY);
        reconnectAttempt++;
        console.log(`Reconnecting in ${delay}ms (attempt ${reconnectAttempt})`);
        setTimeout(connectWebSocket, delay);
    });
}

connectWebSocket();
Why: Replit deployments get a fresh outbound IP on every restart. Transient network issues can drop the Deriv connection. Exponential backoff ensures automatic recovery.
