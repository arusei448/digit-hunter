## ⚠️ Critical Design Principle — Read Before Building

Deriv's volatility indices are generated by an **audited CSPRNG (cryptographically secure pseudo-random number generator)**. They are a **memoryless random walk**. Independent research has rigorously tested this — including a 15-million-tick study and a developer who discovered their "74% accurate" model was actually just a look-ahead data leak (once fixed, accuracy fell to exactly 50%). 

**This means:**

- The AI Analyser must **NOT** claim to predict the next tick's direction.
- It must **NOT** show a "Buy Now" signal with a confidence percentage.
- It **MUST** be a **descriptive market scanner** — it ranks markets by **current statistical conditions** (trend strength, volatility regime, digit skew, momentum), so a trader can see which market currently suits their chosen strategy.

**Every metric shown to the user must be a real, computable number** — no black-box "AI says buy." If the numbers say "no market is trending," the UI must say that honestly.

### 4. Feature Extraction Functions

Add these functions above the HTTP server definition:

```javascript
// ---------- AI Analyser: Feature Extraction ----------

function calculateRSquared(quotes) {
    const n = quotes.length;
    if (n < 5) return 0;
    let sumX = 0, sumY = 0, sumXY = 0, sumX2 = 0, sumY2 = 0;
    for (let i = 0; i < n; i++) {
        sumX += i;
        sumY += quotes[i];
        sumXY += i * quotes[i];
        sumX2 += i * i;
        sumY2 += quotes[i] * quotes[i];
    }
    const num = n * sumXY - sumX * sumY;
    const den = Math.sqrt((n * sumX2 - sumX * sumX) * (n * sumY2 - sumY * sumY));
    if (den === 0) return 0;
    return (num / den) ** 2;
}

function calculateRiseFallRatio(trends) {
    const n = trends.length;
    if (n === 0) return { up: 0, down: 0, upPct: 50, downPct: 50 };
    let up = 0, down = 0;
    for (const t of trends) {
        if (t === 'up') up++;
        else if (t === 'down') down++;
    }
    return {
        up, down,
        upPct: (up / n) * 100,
        downPct: (down / n) * 100
    };
}

function calculateDigitDistribution(digits) {
    const counts = new Array(10).fill(0);
    for (const d of digits) {
        if (d >= 0 && d <= 9) counts[d]++;
    }
    const n = digits.length;
    const percentages = counts.map(c => n > 0 ? (c / n) * 100 : 0);

    // Chi-square test vs uniform
    let chiSq = 0;
    const expected = n / 10;
    if (expected > 0) {
        for (let i = 0; i < 10; i++) {
            chiSq += Math.pow(counts[i] - expected, 2) / expected;
        }
    }
    return { counts, percentages, chiSq };
}

function calculateVolatility(quotes) {
    const n = quotes.length;
    if (n < 5) return 0;
    const returns = [];
    for (let i = 1; i < n; i++) {
        if (quotes[i - 1] !== 0) {
            returns.push((quotes[i] - quotes[i - 1]) / quotes[i - 1]);
        }
    }
    if (returns.length < 3) return 0;
    const mean = returns.reduce((a, b) => a + b, 0) / returns.length;
    const variance = returns.reduce((a, r) => a + Math.pow(r - mean, 2), 0) / returns.length;
    return Math.sqrt(variance);
}

function calculateHurstExponent(quotes) {
    // Simplified R/S approximation over recent 100 ticks
    const series = quotes.slice(-100);
    const n = series.length;
    if (n < 20) return 0.5;

    const mean = series.reduce((a, b) => a + b, 0) / n;
    const deviations = series.map(v => v - mean);
    let cumDev = 0;
    let minDev = 0, maxDev = 0;
    for (const d of deviations) {
        cumDev += d;
        if (cumDev < minDev) minDev = cumDev;
        if (cumDev > maxDev) maxDev = cumDev;
    }
    const R = maxDev - minDev;
    const S = Math.sqrt(deviations.reduce((a, d) => a + d * d, 0) / n);
    if (S === 0) return 0.5;
    const H = Math.log(R / S) / Math.log(n);
    // Clamp to sane range
    return Math.max(0, Math.min(1, H));
}

function calculateStreak(trends) {
    if (trends.length === 0) return { currentStreak: 0, direction: 'none' };
    const last = trends[trends.length - 1];
    if (last === 'flat') return { currentStreak: 0, direction: 'flat' };
    let count = 0;
    for (let i = trends.length - 1; i >= 0; i--) {
        if (trends[i] === last) count++;
        else break;
    }
    return { currentStreak: count, direction: last };
}

// ---------- AI Analyser: Scoring Engine ----------

function scoreMarketForTrendFollowing(features) {
    // Higher score = better trend-following conditions
    let score = 0;

    // Trend strength (R²) — weight 40%
    score += features.rSquared * 40;

    // Rise/Fall imbalance — weight 25% (deviation from 50/50)
    const imbalance = Math.abs(features.riseFall.upPct - 50) / 50;
    score += imbalance * 25;

    // Hurst exponent — weight 20% (> 0.5 = trending)
    if (features.hurst > 0.5) {
        score += ((features.hurst - 0.5) / 0.5) * 20;
    }

    // Volatility — weight 15% (higher vol = more movement to catch, but cap it)
    const volScore = Math.min(features.volatility * 10000, 1);
    score += volScore * 15;

    return Math.round(score);
}

function scoreMarketForMeanReversion(features) {
    let score = 0;

    // Hurst exponent — lower = more mean-reverting (weight 40%)
    if (features.hurst < 0.5) {
        score += ((0.5 - features.hurst) / 0.5) * 40;
    }

    // Digit skew (chi-square) — higher = more non-uniform (weight 30%)
    // Normalize chi-square roughly
    const skewScore = Math.min(features.digitDistribution.chiSq / 20, 1);
    score += skewScore * 30;

    // Trend strength — LOW R² is good for mean reversion (weight 20%)
    score += (1 - features.rSquared) * 20;

    // Streak length — longer streaks = more likely to reverse (weight 10%)
    const streakScore = Math.min(features.streak.currentStreak / 15, 1);
    score += streakScore * 10;

    return Math.round(score);
}

function analyzeAllMarkets() {
    const results = [];
    for (const sym of ALL_MARKETS) {
        const buf = marketBuffers[sym];
        if (!buf || buf.quotes.length < 20) continue;

        const features = {
            symbol: sym,
            name: MARKETS[sym],
            tickCount: buf.quotes.length,
            currentPrice: buf.quotes[buf.quotes.length - 1],
            rSquared: calculateRSquared(buf.quotes),
            riseFall: calculateRiseFallRatio(buf.trends),
            digitDistribution: calculateDigitDistribution(buf.digits),
            volatility: calculateVolatility(buf.quotes),
            hurst: calculateHurstExponent(buf.quotes),
            streak: calculateStreak(buf.trends)
        };

        features.trendScore = scoreMarketForTrendFollowing(features);
        features.reversionScore = scoreMarketForMeanReversion(features);

        results.push(features);
    }

    // Sort by trendScore descending (default view)
    results.sort((a, b) => b.trendScore - a.trendScore);
    return results;
}
```
