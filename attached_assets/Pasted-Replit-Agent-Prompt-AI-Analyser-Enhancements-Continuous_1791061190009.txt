// ---------- New Rise/Fall Strategy Metrics ----------

function calculateEMA(values, period) {
    if (values.length === 0) return 0;
    const k = 2 / (period + 1);
    let ema = values[0];
    for (let i = 1; i < values.length; i++) {
        ema = values[i] * k + ema * (1 - k);
    }
    return ema;
}

function calculateRSI(quotes, period = 14) {
    // RSI on tick-to-tick price changes
    if (quotes.length < period + 1) return 50;
    let gains = 0, losses = 0;
    for (let i = quotes.length - period; i < quotes.length; i++) {
        const diff = quotes[i] - quotes[i - 1];
        if (diff > 0) gains += diff;
        else losses -= diff;
    }
    const avgGain = gains / period;
    const avgLoss = losses / period;
    if (avgLoss === 0) return 100;
    const rs = avgGain / avgLoss;
    return 100 - (100 / (1 + rs));
}

function calculateTickVelocity(timestamps) {
    // Ticks per second over the last 10 ticks
    const n = Math.min(10, timestamps.length);
    if (n < 2) return 0;
    const span = timestamps[timestamps.length - 1] - timestamps[timestamps.length - n];
    if (span <= 0) return 0;
    return n / span;
}

function calculateWeightedPressure(quotes) {
    // EMA-weighted price change — measures directional pressure
    if (quotes.length < 5) return 0;
    const recent = quotes.slice(-20);
    let weighted = 0;
    let weightSum = 0;
    for (let i = 1; i < recent.length; i++) {
        const change = recent[i] - recent[i - 1];
        const weight = i; // newer changes weigh more
        weighted += change * weight;
        weightSum += weight;
    }
    return weightSum > 0 ? weighted / weightSum : 0;
}

function calculateBollingerPosition(quotes, period = 20) {
    // Where is current price relative to Bollinger Bands?
    // Returns -1 (below lower), 0 (middle), +1 (above upper)
    if (quotes.length < period) return 0;
    const recent = quotes.slice(-period);
    const mean = recent.reduce((a, b) => a + b, 0) / period;
    const variance = recent.reduce((a, q) => a + Math.pow(q - mean, 2), 0) / period;
    const stdDev = Math.sqrt(variance);
    if (stdDev === 0) return 0;
    const current = quotes[quotes.length - 1];
    const zScore = (current - mean) / stdDev;
    if (zScore > 1.5) return 1;
    if (zScore < -1.5) return -1;
    return 0;
}

function calculateRunLengthStats(trends) {
    // Average run length and max run length
    if (trends.length < 5) return { avgRun: 0, maxRun: 0, runsCount: 0 };
    let runs = [];
    let currentRun = 1;
    for (let i = 1; i < trends.length; i++) {
        if (trends[i] === trends[i - 1] && trends[i] !== 'flat') {
            currentRun++;
        } else {
            if (currentRun > 0) runs.push(currentRun);
            currentRun = 1;
        }
    }
    runs.push(currentRun);
    const avgRun = runs.reduce((a, b) => a + b, 0) / runs.length;
    const maxRun = Math.max(...runs);
    return { avgRun, maxRun, runsCount: runs.length };
}

function calculateAutocorrelation(quotes) {
    // Lag-1 autocorrelation — measures mean-reversion tendency
    // Positive = trending, negative = mean-reverting
    const n = quotes.length;
    if (n < 10) return 0;
    const returns = [];
    for (let i = 1; i < n; i++) {
        returns.push(quotes[i] - quotes[i - 1]);
    }
    const mean = returns.reduce((a, b) => a + b, 0) / returns.length;
    let num = 0, den = 0;
    for (let i = 1; i < returns.length; i++) {
        num += (returns[i] - mean) * (returns[i - 1] - mean);
    }
    for (let i = 0; i < returns.length; i++) {
        den += Math.pow(returns[i] - mean, 2);
    }
    return den > 0 ? num / den : 0;
}

function scoreMarketForTrendFollowing(features) {
    let score = 0;

    // 1. Trend strength (R²) — 25%
    score += features.rSquared * 25;

    // 2. Rise/Fall imbalance — 15% (deviation from 50/50)
    const imbalance = Math.abs(features.riseFall.upPct - 50) / 50;
    score += imbalance * 15;

    // 3. Hurst exponent — 15% (> 0.5 = trending)
    if (features.hurst > 0.5) {
        score += ((features.hurst - 0.5) / 0.5) * 15;
    }

    // 4. RSI alignment — 15% (strong directional move = RSI away from 50)
    const rsiDeviation = Math.abs(features.rsi - 50) / 50;
    score += rsiDeviation * 15;

    // 5. Positive autocorrelation — 15% (momentum persists)
    if (features.autocorrelation > 0) {
        score += Math.min(features.autocorrelation * 2, 1) * 15;
    }

    // 6. Weighted pressure — 10% (strong directional bias)
    const pressureStrength = Math.min(Math.abs(features.weightedPressure) * 100, 1);
    score += pressureStrength * 10;

    // 7. Bollinger position — 5% (near band = trend continuation)
    score += Math.abs(features.bollingerPosition) * 5;

    return Math.round(score);
}

function scoreMarketForMeanReversion(features) {
    let score = 0;

    // 1. Hurst exponent — 20% (lower = more mean-reverting)
    if (features.hurst < 0.5) {
        score += ((0.5 - features.hurst) / 0.5) * 20;
    }

    // 2. Negative autocorrelation — 25% (strong mean-reversion signal)
    if (features.autocorrelation < 0) {
        score += Math.min(Math.abs(features.autocorrelation) * 2, 1) * 25;
    }

    // 3. RSI near extremes — 15% (>70 or <30 = overbought/oversold)
    const rsiExtreme = Math.max(0, (features.rsi - 70) / 30) + Math.max(0, (30 - features.rsi) / 30);
    score += Math.min(rsiExtreme, 1) * 15;

    // 4. Digit skew (chi-square) — 15%
    const skewScore = Math.min(features.digitDistribution.chiSq / 20, 1);
    score += skewScore * 15;

    // 5. Long streaks — 10% (streaks are more likely to reverse)
    const streakScore = Math.min(features.streak.currentStreak / 15, 1);
    score += streakScore * 10;

    // 6. Bollinger extreme — 10% (price at band = reversal candidate)
    score += Math.abs(features.bollingerPosition) * 10;

    // 7. Low R² — 5% (weak trend = ranging)
    score += (1 - features.rSquared) * 5;

    return Math.round(score);
}

PART 3 — Fix Rise & Fall Stream: Continuous Updates, No Pause
Problem: The tick direction stream on the Rise & Fall page visibly pauses ("thinking") between updates instead of flowing continuously.

Root cause: Three issues combine:

SSE backpressure — the server writes to the client on every tick, but if the client is slow, the Node.js socket buffer fills and the write becomes async, causing visible batching.

DOM thrash — renderStream() reassigns bar.style.height on 40 elements every tick, forcing layout recalculation.

No CSS transition continuity — the browser may drop in-between frames if the change happens during a repaint.

Fix A — Backend: Debounce SSE Writes to ~60ms
In index.js, find where the SSE broadcasts happen (inside ws.on('message')). Replace the direct client.write(payload) loop with a batched broadcast queue:

javascript
// Add near the top of index.js, with the other state:
let directionBroadcastPending = false;
let directionBuffer = [];

function queueDirectionBroadcast(tick) {
    directionBuffer.push(tick);
    if (directionBroadcastPending) return;
    directionBroadcastPending = true;
    setTimeout(() => {
        const batch = directionBuffer;
        directionBuffer = [];
        directionBroadcastPending = false;
        if (batch.length === 0) return;
        // Send only the latest tick (batching collapses duplicates)
        const latest = batch[batch.length - 1];
        const payload = `data: ${JSON.stringify(latest)}\n\n`;
        for (const client of directionClients) {
            try { client.write(payload); } catch (e) { directionClients.delete(client); }
        }
    }, 60);
}
